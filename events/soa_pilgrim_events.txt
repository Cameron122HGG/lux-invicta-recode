	trigger = { 
		has_character_flag = activated_pilgrimage
	}

		clr_character_flag = pilgrimage_to_jerusalem
		clr_character_flag = pilgrimage_to_koln
		clr_character_flag = pilgrimage_to_rome
		clr_character_flag = pilgrimage_to_santiago
		clr_character_flag = pilgrimage_to_constantinople
		clr_character_flag = pilgrimage_to_axum
		clr_character_flag = pilgrimage_to_antioch
        clr_character_flag = pilgrimage_to_carthage
		clr_character_flag = pilgrimage_to_vannes
		clr_character_flag = abominable_pilgrimage
		clr_character_flag = do_not_disturb
		clr_character_flag = preparing_pilgrimage
		clr_character_flag = activated_pilgrimage
		hidden_tooltip = { remove_trait = on_pilgrimage }
	}

	option = {
		name = EVTOPTA_SoA_5000
		piety = 20
		set_character_flag = preparing_pilgrimage
		clr_character_flag = activated_pilgrimage
		hidden_tooltip = {
			if = {
				limit = {
					OR = {
						religion = orthodox  # Jerusalem, Antioch, Axum
						is_heresy_of = orthodox
						religion = nestorian
						is_heresy_of = nestorian
						religion = audianist
						religion = marcionist
						religion = sethian
						religion = sethian_reformed
						religion = monophysite
						is_heresy_of = monophysite
						religion = solomonic_christ
					}
				}
				character_event = { id = SoA.5011 }
			}
			if = {
				limit = {
					OR = {
						religion = catholic # All Catholic holy places
						is_heresy_of = catholic
						religion = cathar_reformed
						religion = christian_populist_reformed
						religion = unctionist
						religion = pelagian
						religion = semipelagian
					}
				}
				character_event = { id = SoA.5020 }
			}
			if = {
				limit = {
					OR = {
						religion = celtic_christ # Vannes
						religion = celtic_christ_reformed
					}
				}
				character_event = { id = lux_soa.3 }
			}
			if = {
				limit = {
					OR = {
						religion = arian # Carthage + Jerusalem
						is_heresy_of = arian
						religion = mari_christ_reformed
						religion = priscillian_reformed
					}
				}
				character_event = { id = lux_soa.6 }
			}
		}
	}
}
