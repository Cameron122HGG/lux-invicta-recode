################################
# Steam Achievement Events
################################

namespace = steam



















# Casanova
character_event = {
	id = steam.019

	is_triggered_only = yes # wol_seduction_events.txt
	hide_window = yes

	ai = no

	trigger = {
		is_ironman = yes
		NOT = { has_character_flag = achievement_casanova }
	}

	immediate = {
		if = {
			limit = {
				check_variable = {
					which = casanova_check
					value < 10
				}
			}

			change_variable = {
				which = casanova_check
				value = 1
			}
		}
		else = {
			set_character_flag = achievement_casanova
		}
	}
}
