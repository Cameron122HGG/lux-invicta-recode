namespace = lidebug

character_event = {
	id = lidebug.1
	hide_window = yes
	is_triggered_only = yes # on_startup

	immediate = {
		log = "LI DEBUG province0: scan started"
		any_character = {
			limit = { is_ruler = yes }
			any_demesne_title = {
				limit = {
					OR = {
						tier = BARON
						tier = COUNT
					}
					NOT = { location = { always = yes } }
				}
				log = "LI DEBUG province0: holder [Prev.GetBestName] ([Prev.GetID]) holds [This.GetName] - no province"
			}
		}
		log = "LI DEBUG province0: scan finished"
	}
}
