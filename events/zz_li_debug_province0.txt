# LI: TEMPORARY DEBUG (2026-11-14) - find rulers whose demesne holds a title with no valid province ("province 0").
# Remove this file and common/on_actions/zz_li_debug_province0.txt once the culprits are known.

namespace = lidebug

character_event = {
	id = lidebug.1
	hide_window = yes
	is_triggered_only = yes # on_startup, on_yearly_pulse

	trigger = { # once per game start, then at most every 2 years (the on_actions fire for every character)
		OR = {
			NOT = { has_global_flag = lidebug_p0 }
			had_global_flag = { flag = lidebug_p0 days = 730 }
		}
	}

	immediate = {
		clr_global_flag = lidebug_p0
		set_global_flag = lidebug_p0
		log = "LI DEBUG province0: scan started"
		any_character = {
			limit = { is_ruler = yes }
			any_demesne_title = {
				limit = {
					OR = {
						tier = BARON
						tier = COUNT
					}
					NOT = { location = { always = yes } }
				}
				log = "LI DEBUG province0: holder [Prev.GetBestName] ([Prev.GetID]) holds [This.GetName] - no province"
			}
		}
		log = "LI DEBUG province0: scan finished"
	}
}
