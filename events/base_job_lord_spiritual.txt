	id = 39653

	picture = GFX_evt_bishop
	border = GFX_event_normal_frame_religion

	has_job_title = yes
	has_character_flag = preacher_arrives

	trigger = {
		has_job_action = action_inquisition
		in_seclusion = no
		NOT = { has_character_modifier = bedridden_illness }

		location = {
			owner = {
				religion_group = pagan_group

				NOR = {
					religion = ROOT
					same_realm = ROOT
					character = ROOT
					is_liege_of = ROOT
				}
			}
		}

		any_playable_ruler = {
			job_spiritual = {
				at_location = ROOT
				has_job_action = action_inquisition
				has_character_flag = preacher_arrives

				NOR = {
					character = ROOT
					same_liege = ROOT
				}
			}
		}
	}

	mean_time_to_happen = {
		years = 10
	}

	option = {
		name = EVTOPTA39653

		random_playable_ruler = {
			job_spiritual = {
				if = {
					limit = {
						at_location = ROOT
						has_job_action = action_inquisition
						has_character_flag = preacher_arrives
						NOT = { character = ROOT }
					}

					character_event = {
						id = 39611 # continues in religious_events.txt
						days = 1
					}
				}
			}
		}
	}
}
