character_event = {
	id = Plus.571

	is_triggered_only = yes # on_yearly_pulse
	hide_window = yes

	only_independent = yes
	has_dlc = "Legacy of Rome"

	trigger = {
		trigger_if = {
			limit = {
				primary_title = {
					has_title_flag = eastern_roman_successor
					real_tier = EMPEROR
				}
			}
			NOT = { has_global_flag = east_rome_lives }
		}
		trigger_else_if = {
			limit = {
				primary_title = {
					has_title_flag = western_roman_successor
					real_tier = EMPEROR
				}
			}
			NOT = { has_global_flag = west_rome_lives }
		}
		trigger_else = {
			primary_title = { title = e_roman_empire }
			NOT = { has_global_flag = rome_restored }
		}
	}

	immediate = {
		if = {
			limit = {
				primary_title = { has_title_flag = eastern_roman_successor }
				NOT = { has_global_flag = east_rome_lives }
			}
			set_global_flag = east_rome_lives
		}
		else_if = {
			limit = {
				primary_title = { has_title_flag = western_roman_successor }
				NOT = { has_global_flag = west_rome_lives } # LI RE: AIO checked east_rome_lives here
			}
			set_global_flag = west_rome_lives
		}
		else_if = {
			limit = {
				primary_title = { title = e_roman_empire }
				NOT = { has_global_flag = rome_restored }
			}
			set_global_flag = rome_restored
		}
	}
}

# LI RE: at game start both Romes exist in 1066 - the Eastern (e_byzantium) and Western (e_roman, S.P.Q.R.) successors
character_event = {
	id = LIprov.1

	is_triggered_only = yes # on_startup (fires for every character: global-flag guard)
	hide_window = yes

	trigger = {
		NOT = { has_global_flag = li_roman_successors_set }
	}

	immediate = {
		set_global_flag = li_roman_successors_set
		log = "LIprov.1: marking the Roman successors (e_byzantium = Eastern, e_roman = Western)"

		e_byzantium = {
			if = {
				limit = { holder_scope = { real_tier = EMPEROR } }
				set_title_flag = eastern_roman_successor
				set_global_flag = east_rome_lives
			}
		}
		e_roman = {
			if = {
				limit = { holder_scope = { real_tier = EMPEROR } }
				set_title_flag = western_roman_successor
				set_global_flag = west_rome_lives
			}
		}
	}
}

# LI RE: Restore the Imperium (user, 2026-11-15). Yearly (and at start) for Byzantium / S.P.Q.R. before the imperium is
# claimed: li_imperium_share = percent of the empire's de jure counties inside the holder's realm.
character_event = {
	id = LIprov.3

	is_triggered_only = yes # on_startup, on_yearly_pulse
	hide_window = yes

	trigger = {
		is_ruler = yes
		primary_title = {
			OR = {
				title = e_byzantium
				title = e_roman
			}
			NOT = { has_title_flag = li_roman_imperium }
		}
	}

	immediate = {
		set_variable = { which = li_imperium_held value = 0 }
		set_variable = { which = li_imperium_total value = 0 }
		primary_title = {
			any_de_jure_vassal_title = {
				limit = { tier = COUNT }
				ROOT = { change_variable = { which = li_imperium_total value = 1 } }
			}
			any_de_jure_vassal_title = {
				limit = {
					tier = COUNT
					holder_scope = {
						OR = {
							character = ROOT
							is_vassal_or_below_of = ROOT
						}
					}
				}
				ROOT = { change_variable = { which = li_imperium_held value = 1 } }
			}
		}
		set_variable = { which = li_imperium_share which = li_imperium_held }
		multiply_variable = { which = li_imperium_share value = 100 }
		if = {
			limit = { check_variable = { which = li_imperium_total value >= 1 } }
			divide_variable = { which = li_imperium_share which = li_imperium_total }
		}
	}
}
