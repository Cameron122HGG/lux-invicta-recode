landless_pope_effect = {
	if = {
		limit = {
			has_landed_title = k_papal_state
			NOT = {
				catholic = {
					is_name_randomized = yes
				}
			}
		}
		if = {
			limit = {
				NOR = {
					any_demesne_province = {
						always = yes
					}
					any_realm_lord = {
						any_demesne_province = {
							always = yes
						}
					}
				}
			}
			#character_event = { id = HFP.40034 }
			#character_event = { id = HFP.40035 }
		}
	}
}
