	intrigue_plot = yes

	#		has_character_flag = allow_plotting
	#	}

	# Target scope
	target_potential = {
		OR = {
			FROM = { is_foe = ROOT }

			AND = {
				OR = {
					has_plot = plot_kill_character
					has_plot = plot_kill_spouse
					has_plot = plot_take_revenge

			has_opinion_modifier = {

		}

		NOT = {
			has_opinion_modifier = {

		}

	type = liege_titles
	military_plot = yes

	# ROOT is plotter
	potential = {

		independent = no

		OR = {
			trigger_if = {
				limit = { crownlaw_title = { is_law_potential = crown_authority_0 } }

				crownlaw_title = {

						has_law = crown_authority_0

			liege = {

	}

	# Title scope
	# FROM is plotter
	allow = {
		holder_scope = {
			NOR = {
				is_married = FROM
				reverse_has_truce = FROM
			}
		}

		OR = {
			FROM = {
				OR = {
					is_feudal = yes
					is_tribal = yes
				}
			}

			AND = {
				FROM = { is_feudal = no }
				lower_real_tier_than = KING

		}

		OR = {

			AND = {
				is_primary_holder_title = yes
				FROM = { is_pretender = yes }
			}

			AND = {
				# Secondary title of highest held tier
				is_primary_holder_title = no
