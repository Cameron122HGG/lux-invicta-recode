check_if_crusader_trigger = {
	conditional_tooltip = {
		trigger = {
			ROOT = {
				religion = catholic
				OR = {
					has_pledged_crusade_participation = yes
					any_war = {
						using_cb = new_crusade
					}
				}
			}
		}
		custom_tooltip = {
			text = crusade_invalidate_other_wars_TT

			NOT = {
				AND = {
					ROOT = {
						religion = catholic
						has_pledged_crusade_participation = yes
					}
					catholic = {
						OR = {
							is_preparing_crusade = yes
							ROOT = {
								any_war = {
									using_cb = new_crusade
								}
							}
						}
					}
					FROM = {
						religion = catholic
						OR = {
							has_landed_title = k_papal_state
							has_pledged_crusade_participation = yes
						}
					}
				}
			}
		}
	}
}
