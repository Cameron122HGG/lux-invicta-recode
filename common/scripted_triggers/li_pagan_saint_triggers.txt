# LI: which saintly bloodline family a pagan saint of each Lux Invicta reformed faith founds (Holy Fury sainthood,
# phase 8 part 2, CLEANSLATE_PLAN.md). Holy Fury only had branches for its 10 vanilla reformed faiths;
# HF_sainthood_events.txt now checks these triggers instead. Whether a faith can have saints at all is decided by the
# religion_beatification feature (chosen at reformation), as in vanilla.

li_saints_germanic_trigger = { # Germanic saintly bloodlines
	OR = {
		religion = norse_pagan_reformed
		religion = german_pagan_reformed
		religion = ragnarok_reformed
		religion = teiwaz_reformed
		religion = norse_christ_reformed
		religion = northern_sun_reformed
		# Celtic faiths: vanilla has no Celtic saintly bloodlines; they use the Germanic ones for now (user decision 2026-11-14)
		religion = celtic_pagan_reformed
		religion = iberian_pagan_reformed
		religion = celtic_christ_reformed
	}
}
