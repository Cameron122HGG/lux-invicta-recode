assassins_bud_mahayana_appropriate_target_trigger = {
	is_within_diplo_range = ROOT
	is_adult = yes
	is_quest_target = no

	OR = {
		is_ruler = no
		AND = {
			ROOT = {
				society_rank = {
					rank = 3
					society = the_assassins_bud_mahayana
				}
			}
			primary_title = { higher_real_tier_than = COUNT }
		}
		AND = {
			ROOT = {
				society_rank = {
					rank = 2
					society = the_assassins_bud_mahayana
				}
			}
			primary_title = { lower_real_tier_than = EMPEROR }
		}
		AND = {
			ROOT = {
				society_rank = {
					rank = 1
					society = the_assassins_bud_mahayana
				}
			}
			primary_title = { lower_real_tier_than = DUKE }
		}
	}
}
