# Pagans can subjugate other pagans
# LI RE: Lux Invicta's version (from master), not CleanSlate's - it uses LI's own world.
pagan_subjugation = {
	name = CB_NAME_PAGAN_SUBJUGATION
	war_name = WAR_NAME_PAGAN_SUBJUGATION
	sprite = 8
	truce_days = 365
	hostile_against_others = yes
	is_permanent = yes
	can_ask_to_join_war = yes
	check_de_jure_tier = KING # this scans all de jure kingdoms for the counties which are held by or vassals(or below) of selected character. Only valid if is_permanent = yes
	allowed_to_target_tributaries = no
	# allowed_to_target_suzerains = no

	can_use = {
		ROOT = {
			government = tribal_government
			has_horde_culture = no # Should use tribal invasion instead
			mercenary = no
			is_theocracy = no
			OR = {
				AND = {
					has_dlc = "Horse Lords"
					is_nomadic = no
				}
				AND = {
					NOT = {
						has_dlc = "Horse Lords"
					}
					has_horde_culture = no
				}
			}
			OR = {
				NOT = {
					has_character_modifier = launched_subjugation
				}
				AND = {
					has_ambition = obj_become_king
					capital_scope = {
						kingdom = {
							FROM = {
								any_realm_title = {
									kingdom = {
										title = PREVPREVPREV
									}
								}
							}
						}
					}
				}
			}
			NOR = {
				same_realm = FROM
				is_liege_or_above = FROM
			}
			custom_tooltip = {
				text = NOT_BUSY
				hidden_tooltip = {
					is_inaccessible_trigger = no
				}
			}
			clanic_trigger = yes
		}
		FROM = {
			in_revolt = no
			is_nomadic = no
			culture = ROOT
			OR = {
				is_reformed_religion = no
				is_reformed_religion = ROOT
				religion_group = ROOT
			}
			NOT = {
				any_war = {
					defender = { character = FROM }
					OR = {
						using_cb = pagan_subjugation
						using_cb = nomad_subjugation
					}
				}
			}
		}
	}

	can_use_gui = {
		ROOT = {
		prestige = 100
		}
	}

	on_add = {
		ROOT = {
			add_character_modifier = {
				name = launched_subjugation
				years = 100
				hidden = yes
			}
			# scaled_wealth = -0.5
			# wealth = -50
			prestige = -200
		}
	}
	on_add_title = {
		ROOT = {
			if = {
				limit = {
					OR = {
						NOT = { has_ambition = obj_become_king }
						capital_scope = {
							kingdom = {
								NOT = { title = PREVPREVPREV }
							}
						}
					}
				}
				add_character_modifier = {
					name = launched_subjugation
					days = 3650
					hidden = yes
				}
			}
		}
	}

	on_success = {
		ROOT = {
			prestige = 150
			piety = 10
		}
		#any_attacker = {
		#	limit = { character = ROOT }
		#	participation_scaled_prestige = 100
		#}
		any_attacker = {
			limit = { NOT = { character = ROOT } }
			hidden_tooltip = { participation_scaled_prestige = 100 }
		}
		FROM = {
			#piety = -200
			prestige = -200

			if = { # Reformed pagans are forced to convert back
				limit = {
					is_reformed_religion = ROOT
					controls_religion = no
				}
				religion = ROOT
			}
			
			custom_tooltip = {
				text = tribal_subjugation_tip
			}
				
			hidden_tooltip = {
				any_demesne_title = {
					limit = {
						higher_tier_than = ROOT
					}
					ROOT = { grant_title_no_opinion = { target = PREV type = invasion } }
					add_pressed_claim = FROM
				}
				any_demesne_title = {
					limit = {
						tier = ROOT
					}
					ROOT = { grant_title_no_opinion = { target = PREV type = invasion } }
					add_pressed_claim = PREV
				}
				set_defacto_liege = ROOT
			}
		}
	}

	on_fail = {
		FROM = {
			prestige = 100
		}
		any_defender = {
			limit = { character = FROM }
			participation_scaled_prestige = 75
		}
		any_defender = {
			limit = { NOT = { character = FROM } }
			hidden_tooltip = { participation_scaled_prestige = 75 }
		}
		ROOT = {
			prestige = -200
			transfer_scaled_wealth = {
				to = FROM
				value = 0.5
			}
		}
	}

	on_invalidation = {
		if = {
			limit = {
				NOT = {
					has_opinion_modifier = {
						modifier = declared_independence_nomad
						who = FROM
					}
				}
			}
			ROOT = {
				remove_character_modifier = launched_subjugation
			}
		}
	}

	on_reverse_demand = {
		ROOT = {
			prestige = -500
			wealth = -100
			transfer_scaled_wealth = {
				to = FROM
				value = 5.0
			}
		}
		FROM = {
			piety = 50
			wealth = 100
			prestige = 300
		}
		FROM = {
			if = {
				limit = {
					has_dlc = "Legacy of Rome"
					OR = {
						has_landed_title = e_byzantium
						has_landed_title = e_roman
						has_landed_title = e_roman_empire
						AND = {
							OR = {
								culture = roman
								culture = romanobrython
								culture = romanogothic
								culture = galloroman
								culture = greek
							}
							independent = yes
							higher_tier_than = duke
						}
					}
				}
				hidden_tooltip = { character_event = { id = LoR.30 days = 12 } }
			}
		}
		any_defender = {
			limit = { character = FROM }
			participation_scaled_prestige = 100
		}
		any_defender = {
			limit = { NOT = { character = FROM } }
			hidden_tooltip = { participation_scaled_prestige = 100 }
		}
	}

	attacker_ai_victory_worth = {
		factor = 200
	}

	attacker_ai_defeat_worth = {
		factor = 100
	}

	defender_ai_victory_worth = {
		factor = -1 # always accept
	}

	defender_ai_defeat_worth = {
		factor = 100
	}
	ai_will_do = {
		factor = 1
	}
}

			limit = { NOT = { character = ROOT } }
			hidden_tooltip = { participation_scaled_prestige = 300 }
		}

			prestige = -500
		}
	}

		custom_tooltip = {
			text = tribal_invasion_succ_tip

					random_realm_province = {

							de_jure_liege_or_above = PREVPREV

								NOT = { culture = ROOT }
								NOT = { religion = ROOT }
							}
						}
						culture = ROOT
						religion = ROOT
					}
				}

					gain_all_occupied_titles = { who = FROM type = invasion }

					limit = {
						FROM = { is_liege_or_above = ROOT }

									is_adult = yes
									practical_age < 30
									trait = zealous
								}

									religion = attacker
									is_adult = yes
									trait = zealous
								}

								preferred_limit = {
									religion = attacker
									is_adult = yes
								}

								preferred_limit = {
									is_adult = yes
									religion_is_matriarchal_trigger = yes
								}

								preferred_limit = {
									is_adult = yes
								}

								save_event_target_as = new_swapped_ruler
							}
						}
						else = {
							location = {
								create_character = {
									religion = attacker
									culture = defender
									dynasty = random
									random_traits = yes
									female = yes
									age = 23
									health = 7
									fertility = 0.7

									attributes = {
										martial = 8
										diplomacy = 8
										stewardship = 8

								new_character = {
									remove_trait = cynical
									wealth = 250
									prestige = 300
									piety = 100
									save_event_target_as = new_swapped_ruler
								}
							}
						}

						primary_title = { save_event_target_as = title_usurped }

						any_demesne_title = {
							limit = { real_tier = event_target:title_usurped }

								target = event_target:new_swapped_ruler

							}

						if = {
							limit = { religion_is_equal_trigger = no }

								add_law = {

									opinion_effect = no

										law = enatic_succession
										cooldown = no
										opinion_effect = no
									}
								}

									limit = { is_law_potential = enatic_cognatic_succession }

									add_law = {
										law = enatic_cognatic_succession
										cooldown = no
										opinion_effect = no
									}
								}
								else = {
									add_law = {
										law = true_cognatic_succession
										cooldown = no
										opinion_effect = no
									}
								}
							}

							recalc_succession = yes
						}
					}

					if = {

							}

							name = in_non_aggression_pact
							years = 50
							who = event_target:new_swapped_ruler

					limit = {

							has_dlc = "Horse Lords"
							has_dlc = "Jade Dragon"

					limit = { uses_decadence = yes }
					participation_scaled_decadence = -10
				}

					name = won_holy_war
					years = 20
				}
			}
			else = {
				hidden_effect = {
					participation_scaled_piety = 50
					participation_scaled_prestige = 25

					if = {
						limit = { uses_decadence = yes }
						participation_scaled_decadence = -10
					}
				}
			}

			hidden_effect = {
				if = {
					limit = {
						has_ambition = obj_strengthen_religion
						NOT = { religion = defender }
					}

					change_variable = {
						which = strengthen_religion
						value = 1
					}
				}
			}
		}

		defender = {
			show_scope_change = no

			piety = -100
			prestige = -50
		}
	}

	on_fail = {
		attacker = {
			show_scope_change = no

			piety = -125
			prestige = -250
		}
	}

				if = {

					name = lost_holy_war

					}
				}
			}
		}

		any_defender = {

			if = {

				participation_scaled_piety = 200
				participation_scaled_prestige = 100

				if = {
					limit = { uses_decadence = yes }
					participation_scaled_decadence = -10
				}

				if = {
					limit = { religion_is_matriarchal_trigger = no }

					religion_authority = {
						name = won_holy_war
						years = 20
					}
				}
			}
			else = {
				hidden_effect = {
					participation_scaled_piety = 200
					participation_scaled_prestige = 100

					if = {
						limit = { uses_decadence = yes }
						participation_scaled_decadence = -10
					}
				}
			}

			hidden_effect = {
				if = {
					limit = {
						has_ambition = obj_strengthen_religion
						NOT = { religion = attacker }
					}

					change_variable = {
						which = strengthen_religion
						value = 1

			vassal_of = FROM
		}
	}

		# Attacker

			NOT = { religion = FROM }

			independent = yes
			higher_real_tier_than = ROOT
		}
	}

		defender = {
			if = {
				limit = {
					independent = no
					liege = { religion = ROOT }
				}

						any_attacker = {
							set_defacto_liege = PREV
						}
					}
				}
				else = {
					liege = {
						any_attacker = {
							set_defacto_liege = PREV
						}
					}
				}
			}

			else_if = {
				limit = {
					OR = {
						independent = yes
						liege = { religion = ROOT }
					}
				}

				any_attacker = {
					set_defacto_liege = ROOT
					clr_character_flag = flag_truebelievers_revolter
				}
			}

					add_pressed_claim = defender
				}
			}

				participation_scaled_prestige = 100
				participation_scaled_piety = 100
			}
		}

			any_demesne_title = { # All titles
				limit = {
					higher_real_tier_than = COUNT
					NOT = {	has_law = tribal_organization_0	}
				}

				trigger_switch = { # Lower tribal organization
					on_trigger = has_law

					tribal_organization_1 = { add_law = tribal_organization_0 }

				}
			}

	}
