bid_for_independence = {
	name = CB_NAME_INDEPENDENCE
	war_name = WAR_NAME_INDEPENDENCE
	sprite = 12
	truce_days = 3650
	sort_priority = 1500

	is_permanent = yes
	is_independence = yes

	can_use_gui = {
		not_fellow_crusader_trigger = yes
	}

	can_use = {
		# Attacker
		ROOT = {
			vassal_of = FROM
			higher_real_tier_than = BARON
			mercenary = no

			trigger_if = {
				limit = { is_patrician = yes }

				# No independence factions in merchant republics
				is_merchant_republic = yes

				# Can no longer be a merchant republic when becoming
				# a direct vassal of another merchant republic
				liege = {
					trigger_if = {
						limit = { independent = no }
						liege_before_war = { is_merchant_republic = no }
					}
				}
			}

			trigger_if = {
				limit = { is_nomadic = yes }

				NOT = { has_blood_oath_with = FROM }

				clan_opinion = {
					who = FROM
					value <= 0
				}
			}
		}

		# Defender
		FROM = {
			in_revolt = no
			higher_real_tier_than = ROOT

			NOT = { # isn't attacking liege with one of these CBs
				any_war = {
					attacker = {
						character = PREVPREV
					}

					OR = {
						using_cb = bid_for_independence
						using_cb = depose_liege
					}
				}
			}

			# No above liege is revolting
			liege_before_war = {
				in_revolt = no

				liege_before_war = {
					in_revolt = no
				}
			}
		}
	}

	is_valid = {
		# Attacker
		ROOT = {
			liege_before_war = {
				character = FROM
			}

			trigger_if = {
				limit = { is_patrician = yes }

				# No independence factions in merchant republics
				is_merchant_republic = yes

				# Can no longer be a merchant republic when becoming
				# a direct vassal of another merchant republic
				liege_before_war = {
					trigger_if = {
						limit = { independent = no }
						liege_before_war = { is_merchant_republic = no }
					}
				}
			}
		}

		# Defender
		FROM = {
			in_revolt = no
			higher_real_tier_than = ROOT

			# No above liege is revolting
			liege_before_war = {
				in_revolt = no

				liege_before_war = {
					in_revolt = no
				}
			}
		}
	}

	on_add = {
		anger_china_effect = yes
	}

	on_success = {
		any_attacker = {
			show_scope_change = no

			if = {
				limit = { character = attacker }

				defender = {
					if = {
						limit = { independent = yes }
						attacker = { save_event_target_as = new_liege }
					}
					else = {
						liege_before_war = { save_event_target_as = new_liege }
					}
				}

				set_defacto_liege = event_target:new_liege
				prestige = 100
				participation_scaled_prestige = 100

				hidden_effect = {
					any_demesne_title = {
						add_pressed_claim = defender
					}

					set_character_flag = ai_flag_never_transfer
				}
			}
			else = {
				hidden_effect = { participation_scaled_prestige = 100 }
			}
		}

		defender = {
			show_scope_change = no

			# Lower crown authority
			any_demesne_title = {
				limit = {
					is_crown_law_title = yes
					is_law_potential = crown_authority_0
				}

				trigger_switch = {
					on_trigger = has_law

					crown_authority_1 = { add_law = crown_authority_0 }
					crown_authority_2 = { add_law = crown_authority_1 }
					crown_authority_3 = { add_law = crown_authority_2 }
					crown_authority_4 = { add_law = crown_authority_3 }
				}
			}
		}
	}

	on_fail = {
		any_attacker = {
			show_scope_change = no

			if = {
				limit = { character = attacker }

				prestige = -50
				participation_scaled_prestige = -50
			}
			else = {
				hidden_effect = { participation_scaled_prestige = -50 }
			}
		}
	}

	on_reverse_demand = {
		attacker = {
			show_scope_change = no

			if = {
				limit = { character = attacker }

				imprison = defender
				prestige = -100
				participation_scaled_prestige = -100
			}
			else = {
				hidden_effect = { participation_scaled_prestige = -100 }
			}
		}

		defender = {
			show_scope_change = no
			prestige = 50
		}
	}

	on_attacker_leader_death = {
		any_defender = {
			letter_event = { id = 251 } # Our enemy has died, the war has ended
		}

		end_war = invalid
	}

	attacker_ai_victory_worth = {
		factor = -1 # always accept
	}

	attacker_ai_defeat_worth = {
		factor = 150
	}

	defender_ai_victory_worth = {
		factor = -1 # always accept
	}

	defender_ai_defeat_worth = {
		factor = 100
	}
}

# LI: Lux Invicta's version (from master), not CleanSlate's - it uses LI's own world.
depose_liege = {
	name = CB_NAME_DEPOSE_LIEGE
	war_name = WAR_NAME_DEPOSE_LIEGE
	sprite = 12
	truce_days = 3650
	
	is_permanent = yes
	can_call_vassals = no
	major_revolt = no

	can_use = {
		always = no
		ROOT = {
			de_facto_liege = FROM
			custom_tooltip = {
				text = NOT_BUSY
				hidden_tooltip = {
					is_inaccessible_trigger = no
				}
			}
		}
		
		FROM = {
			is_feudal = yes
			primary_title = {
				is_vice_royalty = no
			}
			NOT = {
				any_war = {
					OR = {
						AND = { # Cannot depose liege when our liege is currently involved in such a war
							attacker = {
								character = PREVPREV
							}
							OR = {
								using_cb = bid_for_independence
								using_cb = depose_liege
							}
						}
						AND = { # Cannot depose liege if someone else is already deposing liege
							defender = {
								character = PREVPREV
							}
							using_cb = depose_liege
						}
					}
				}
			}
		}
	}

	is_valid = {
		ROOT = {
			OR = {
				liege = {
					character = PREV # either independent
				}
				liege = { 
					FROM = { 
						is_liege_or_above = PREV # or have shared liege
					}
				}
			}
		}
		FROM = {
			any_demesne_title = {
				NOT = { lower_tier_than = ROOT }
				temporary = no
			}
			in_revolt = no
			liege_before_war = {
				in_revolt = no
				liege_before_war = {
					in_revolt = no
				}
			}
		}
	}

	on_add = {
		hidden_tooltip = {
			FROM = {
				any_vassal = {
					letter_event = { id = addrebels.0 days = 1 }
				}
			}
		}
	}

	on_success = {
		FROM = {
			prestige = -500
			if = {
				limit = { NOT = { has_dlc = "Zeus" } }
				any_demesne_title = { # All titles
					limit = {
						OR = {
							tier = king
							tier = emperor
							is_primary_holder_title = yes
						}
					}
					if = {
						limit = { has_law = centralization_1 }
						add_law = centralization_0
					}
					if = {
						limit = { has_law = centralization_2 }
						add_law = centralization_1
					}
					if = {
						limit = { has_law = centralization_3 }
						add_law = centralization_2
					}
					if = {
						limit = { has_law = centralization_4 }
						add_law = centralization_3
					}
					if = {
						limit = { has_law = clanic_centralization_1 }
						add_law = clanic_centralization_0
					}
					if = {
						limit = { has_law = clanic_centralization_2 }
						add_law = clanic_centralization_1
					}
					if = {
						limit = { has_law = clanic_centralization_3 }
						add_law = clanic_centralization_2
					}
					if = {
						limit = { has_law = clanic_centralization_4 }
						add_law = clanic_centralization_3
					}
				}
			}

			abdicate_to_most_liked_by = ROOT
		}
		any_attacker = {
			limit = { character = ROOT }
			participation_scaled_prestige = 250
		}
		any_attacker = {
			limit = { NOT = { character = ROOT } }
			hidden_tooltip = { participation_scaled_prestige = 250 }
		}
	}

	on_fail = {
		ROOT = {
			prestige = -100
		}
	}

	on_reverse_demand = {
		ROOT = {
			prestige = -200
			prisoner = FROM
		}
		any_defender = {
			limit = { character = FROM }
			participation_scaled_prestige = 50
		}
		any_defender = {
			limit = { NOT = { character = FROM } }
			hidden_tooltip = { participation_scaled_prestige = 50 }
		}
	}

	attacker_ai_victory_worth = {
		factor = -1 # always accept
	}
	
	attacker_ai_defeat_worth = {
		factor = 150
	}

	defender_ai_victory_worth = {
		factor = -1 # always accept
	}
	
	defender_ai_defeat_worth = {
		factor = 100
	}
}

overthrow_ruler = {
	name = CB_NAME_OVERTHROWRULER
	war_name = WAR_NAME_OVERTHROWRULER
	sprite = 12
	truce_days = 3650
	
	is_revolt_cb = yes
	can_call_vassals = yes
	attacker_can_call_allies = no
	major_revolt = yes
	is_tyranny_cb = yes

	can_use = {
		FROM = {
			primary_title = {
				is_vice_royalty = no
			}
		}
		ROOT = {
			de_facto_liege = FROM
			custom_tooltip = {
				text = NOT_BUSY
				hidden_tooltip = {
					is_inaccessible_trigger = no
				}
			}
		}
	}

	is_valid = {
		ROOT = {
			OR = {
				liege = {
					character = PREV # either independent
				}
				liege = { 
					FROM = { 
						is_liege_or_above = PREV # or have shared liege
					}
				}
			}
		}
		FROM = {
			any_demesne_title = {
				NOT = { lower_tier_than = ROOT }
				temporary = no
			}
		}
	}
	on_add = {
		hidden_tooltip = {
			FROM = {
				any_vassal = {
					letter_event = { id = addrebels.0 days = 1 }
				}
			}
		}
	}

	on_success = {
		ROOT = {
			primary_title = {
				add_pressed_claim = FROM	
			}
		}
		FROM = {
			prestige = -500
			remove_opinion = {
				who = ROOT
				modifier = declared_war
			}
			if = {
				limit = { NOT = { has_dlc = "Zeus" } }
				any_demesne_title = { # All titles
					limit = {
						higher_tier_than = duke
	#					is_titular = no
					}
					if = {
						limit = { has_law = centralization_1 }
						add_law = centralization_0
					}
					if = {
						limit = { has_law = centralization_2 }
						add_law = centralization_0
					}
					if = {
						limit = { has_law = centralization_3 }
						add_law = centralization_1
					}
					if = {
						limit = { has_law = centralization_4 }
						add_law = centralization_2
					}
					if = {
						limit = { has_law = clanic_centralization_1 }
						add_law = clanic_centralization_0
					}
					if = {
						limit = { has_law = clanic_centralization_2 }
						add_law = clanic_centralization_0
					}
					if = {
						limit = { has_law = clanic_centralization_3 }
						add_law = clanic_centralization_1
					}
					if = {
						limit = { has_law = clanic_centralization_4 }
						add_law = clanic_centralization_2
					}
				}
			}
			if = {
				limit = { has_dlc = "Zeus" }
				any_demesne_title = { # All titles
					limit = {
						higher_tier_than = count
						NOT = { has_law = sovereignty_title_law_0 }
					}
					if = {
						limit = {
							OR = {
								has_law = sovereignty_title_law_1
								has_law = sovereignty_title_law_2
							}
						}
						add_law = sovereignty_title_law_0
					}
				}
			}
			abdicate = yes
		}
		any_attacker = {
			limit = { character = ROOT }
			participation_scaled_prestige = 250
		}
		any_attacker = {
			limit = { NOT = { character = ROOT } }
			hidden_tooltip = { participation_scaled_prestige = 250 }
		}
	}

	on_fail = {
		ROOT = {
			prestige = -100
		}
		FROM = {
			prestige = -100
			remove_opinion = {
				who = ROOT
				modifier = declared_war
			}
		}
	}

	on_reverse_demand = {
		ROOT = {
			prestige = -200
			prisoner = FROM
		}
		any_defender = {
			limit = { character = FROM }
			participation_scaled_prestige = 50
		}
		any_defender = {
			limit = { NOT = { character = FROM } }
			hidden_tooltip = { participation_scaled_prestige = 50 }
		}
	}

	attacker_ai_victory_worth = {
		factor = -1 # always accept
	}
	
	attacker_ai_defeat_worth = {
		factor = 150
	}

	on_defender_leader_death = {
		hidden_tooltip = {
			any_attacker = {
				letter_event = {
					id = 251
				}
			}
		}
		
		end_war = invalid
	}

	defender_ai_victory_worth = {
		factor = -1 # always accept
	}
	
	defender_ai_defeat_worth = {
		factor = 100
	}
}
