# Loosen clan control 
loosen_clan_control_law = {
	name = CB_NAME_LOOSEN_CLAN_CONTROL_LAW
	war_name = WAR_NAME_LOOSEN_CLAN_CONTROL_LAW
	sprite = 6
	truce_days = 3650
	can_call_allies = no
	can_ask_to_join_war = no
	major_revolt = yes
	
#	attacker_alliance_occ_warscore = yes # Occupying any holding of any attacker counts towards war score
	
	can_use = {
		ROOT = {
			vassal_of = FROM
			custom_tooltip = {
				text = NOT_BUSY
				hidden_tooltip = {
					is_inaccessible_trigger = no
				}
			}
		}
	}
	
	can_use_title = {
		holder_scope = { independent = yes }
		OR = {
			higher_tier_than = duke
			is_primary_holder_title = yes
		}
		OR = {
			has_law = clanic_centralization_1
			has_law = clanic_centralization_2
			has_law = clanic_centralization_3
			has_law = clanic_centralization_4
		}
	}

	is_valid = {
		FROM = {
			in_revolt = no
			liege_before_war = {
				in_revolt = no
				liege_before_war = {
					in_revolt = no
				}
			}
		}
		ROOT = {
			OR = {
				liege = {
					character = PREV # either independent
				}
				liege = { 
					FROM = { 
						is_liege_or_above = PREV # or have shared liege
					}
				}
			}
		}
	}
	on_success = {
		ROOT = {
			opinion = {
				modifier = opinion_loosened_clan_control
				who = FROM
				months = 12
			}
		}
		any_attacker = {
			limit = { character = ROOT }
			participation_scaled_prestige = 100
			hidden_tooltip = {
				disband_event_forces = faction_revolters
			}
		}
		any_attacker = {
			limit = { NOT = { character = ROOT } }
			hidden_tooltip = { participation_scaled_prestige = 100 }
			hidden_tooltip = {
				disband_event_forces = faction_revolters
			}
		}
		FROM = { hidden_tooltip = { disband_event_forces = faction_loyalists } }
	}

	on_success_title = {
		if = {
			limit = { has_law = clanic_centralization_1 }
			add_law_w_cooldown = clanic_centralization_0
		}
		if = {
			limit = { has_law = clanic_centralization_2 }
			add_law_w_cooldown = clanic_centralization_1
		}
		if = {
			limit = { has_law = clanic_centralization_3 }
			add_law_w_cooldown = clanic_centralization_2
		}
		if = {
			limit = { has_law = clanic_centralization_4 }
			add_law_w_cooldown = clanic_centralization_3
		}
	}

	on_fail = {
		ROOT = {
			prestige = -100
		}
		
		any_defender = {
			limit = { character = FROM }
			participation_scaled_prestige = 50
		}
		any_defender = {
			limit = { NOT = { character = FROM } }
			hidden_tooltip = { participation_scaled_prestige = 50 }
		}
		
		hidden_tooltip = {
			any_attacker = {
				disband_event_forces = faction_revolters
				add_character_modifier = {
					name = faction_loosen_clan_control_ultimatum_timer
					hidden = yes
					duration = 3650
				}
			}
		}
		FROM = { hidden_tooltip = { disband_event_forces = faction_loyalists } }
	}

	on_reverse_demand = {
		ROOT = {
			prestige = -200
			prisoner = FROM
		}
		any_defender = {
			limit = { character = FROM }
			participation_scaled_prestige = 100
		}
		any_defender = {
			limit = { NOT = { character = FROM } }
			hidden_tooltip = { participation_scaled_prestige = 100 }
		}
		
		hidden_tooltip = {
			any_attacker = {
				limit = { liege = { character = FROM } }
				prisoner = FROM
				disband_event_forces = faction_revolters
				add_character_modifier = {
					name = faction_loosen_clan_control_ultimatum_timer
					hidden = yes
					duration = 3650
				}
			}
		}
		FROM = { hidden_tooltip = { disband_event_forces = faction_loyalists } }
	}

	attacker_ai_victory_worth = {
		factor = -1 # always accept
	}
	
	attacker_ai_defeat_worth = {
		factor = 100
	}

	defender_ai_victory_worth = {
		factor = -1 # always accept
	}
	
	defender_ai_defeat_worth = {
		factor = 100
	}
}
