	dynastic_meddling = {
		diplomacy_icon = GFX_special_bloodline_decision

		filter = dynasty
		ai_target_filter = dynasty
		ai_check_interval = 40

		only_playable = yes

		from_potential = {
			has_dlc = "Holy Fury"
			is_adult = yes
			prisoner = no
			is_incapable = no

			any_owned_bloodline = {
				has_bloodline_flag = bloodline_dynastic_meddling
				bloodline_is_active_for = PREV
			}
		}

		potential = {
			is_close_relative = FROM
			is_within_diplo_range = FROM
			is_adult = yes
			prisoner = no
			is_incapable = no
		}

		allow = {
			show_only_failed_conditions = yes

			FROM = {
				piety >= 0
				age >= 40 # Meddler must be old

				custom_tooltip = {
					text = tooltip_dynastic_meddling_cooldown
					NOT = { has_character_modifier = dynastic_meddling_cooldown }
				}
			}

			ROOT = {
				age < 35 # Target must be young
				diplomatic_immunity = no
			}
		}

		effect = {
			FROM = {
				trigger_switch = {
					on_trigger = real_tier

					EMPEROR  = { piety = -100 }
					KING     = { piety =  -75 }
					DUKE     = { piety =  -50 }
					COUNT    = { piety =  -30 }
					BARON    = { piety =  -15 }
					fallback = { piety =   -5 }
				}
			}

			custom_tooltip = {
				text = tooltip_dynastic_meddling

				FROM = {
					character_event = { id = HF.24152 } # Family Bloodline - Dynastic meddling

					add_character_modifier = {
						name = dynastic_meddling_cooldown
						years = 5
						hidden = yes
					}
				}

				if = {
					limit = { ai = no }

				#	CleanSlate: unused?
				#	change_variable = {
				#		which = family_nagging
				#		value = 1
				#	}

					# Stop the AI from spamming the player
					add_character_modifier = {
						name = dynastic_meddling_target_cooldown
						years = 5
						hidden = yes
					}
				}
			}
		}

		ai_will_do = {
			factor = 1

			trigger = {
				FROM = {
					trigger_if = {
						limit = { higher_real_tier_than = COUNT }
						piety >= 200
					}

					NOR = {
						trait = shy
						trait = patient
						trait = slothful
						trait = depressed
						trait = content
						is_rival = ROOT
						is_lover = ROOT
						is_friend = ROOT
					}
				}

				# Player should not be spammed
				NOT = { has_character_modifier = dynastic_meddling_target_cooldown }

				# AI only picks decision if host, guest, liege, or spouse of target
				OR = {
					host = { character = FROM }
					FROM = { host = { character = ROOT } }
					is_married = FROM
					is_liege_or_above_of = FROM
				}
			}

			mult_modifier = { # Will interfere
				factor = 5

				FROM = {
					OR = {
						trait = diligent
						trait = wroth
						trait = proud
						trait = just
						trait = gregarious
						trait = socializer
					}
				}
			}

			mult_modifier = { # Needs a good talk
				factor = 3

				ROOT = {
					OR = {
						trait = stressed
						trait = depressed
						trait = drunkard
						trait = slothful

						trigger_if = {
							limit = { trait = lustful }

							trigger_if = {
								limit = { is_married = yes }
								num_of_children >= 2
							}
						}

						has_focus = focus_seduction
						trait = uncouth
						trait = fat
						trait = malnourished
					}
				}
			}

			mult_modifier = {
				factor = 6
				FROM = { has_focus = focus_family }
			}

			mult_modifier = {
				factor = 4
				FROM = { age >= 60 }
				ROOT = { age <  25 }
			}
		}
	}
