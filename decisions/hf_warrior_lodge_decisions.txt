							name = dueling
							value = unrestricted

				# But otherwise: find a target, valid both relationship-, and healthwise...

					custom_tooltip = {

								has_quest = quest_warrior_lodge_duel_honor

				}
			}

		}

		ai_will_do = {
			factor = 0.1

				factor = 1.5
